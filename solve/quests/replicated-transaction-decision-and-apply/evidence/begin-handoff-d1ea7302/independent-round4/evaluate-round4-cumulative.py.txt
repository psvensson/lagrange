def check(label,ok):checks.append((label,bool(ok)))
x=read('original-adversary-new.json')

check('engine getter refused without execution',s['engine-accessor-configuration']['getterCalls']==0 and not s['engine-accessor-configuration']['accepted'])
check('injected spent control turn refuses',not s['control-deadline-completion-before-timer']['accepted'])

check('shutdown census cannot reinstall or rewrite object',s['shutdown-with-inflight-census-and-late-insert']['afterLate']['state'] is None)

check('expired claim reenters recovery',not s['expired-success-claim-never-hands-off']['retry']['success'] and s['expired-success-claim-never-hands-off']['handoffs']>0)
for x in read('advanced-orderings.json'):
 d=x['beforeRetry']['durable']['decision_digest']; check(x['order']+' missing progress redrives exact COMMIT',len(x['commitEmissions'])==1 and not x['rollbackEmissions'] and x['commitEmissions'][0]['identity']['decisionDigest']==d and x['afterReplay']['durable']['decision_digest']==d and x['afterReplay']['durable']['decision']=='COMMIT');check(x['order']+' advanced disposition retains same record and live slot',x['afterReplay']['sameRecord'] and x['afterReplay']['consumerRetained'] and x['afterReplay']['recordCount']==1)

x={r['name']:r for r in read('round3-interactions.json')['results']};r=x['known-advanced-then-insert-applied-replay'];check('B3-1 advanced authority cannot become accepted ACTIVE on old INSERT replay',not r['accepted']);check('B3-1 witness preserves exact INSERT bytes key and durable COMMIT',len(r['writeTrace'])==1 and r['after']['writes']==1 and r['before']['durable']==r['after']['durable'])
for r in x['caller-expiry-live-session-retry-both-orderings']['cases']:check('B3-2 '+r['order']+' live session can consume initial-row result after caller expiry',r['accepted'])
r=x['sweep-preserves-live-success-claim'];check('B3-3 unexpired live claim survives unrelated sweep',r['validBefore'] and r['validAfter'])
r=read('round3-pgwire-sweep.json');check('B3-3 actual pgwire queued sweep preserves BEGIN acceptance',r['messageTypes']==['C','Z'] and r['frames'][-1].endswith('T'))
r=read('round3-resource-probe.json');check('B3-4 canonical envelope covers actual field lower bound',r['observed']['diagnostics']['recordBytes']>=r['observed']['minCanonicalBytes'])
for label,ok in checks:print(('PASS ' if ok else 'FAIL ')+label)
print(str(sum(ok for _,ok in checks))+'/'+str(len(checks))+' current assertions pass; '+str(sum(not ok for _,ok in checks))+' failures in four grouped findings')
