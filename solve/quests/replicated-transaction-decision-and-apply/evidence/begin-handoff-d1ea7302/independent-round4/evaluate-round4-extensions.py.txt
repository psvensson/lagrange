import json,pathlib,sys
p=pathlib.Path(__file__).parent
read=lambda name:json.loads((p/name).read_text())
checks=[]
def check(name,ok):checks.append((name,bool(ok)))
x={r['name']:r for r in read('round4-lifecycle-probes.json')['results']}
r=x['advanced-recovery-after-original-control-expiry']
check('B4-1a advanced retry uses current pre-FIFO turn budget',r['retained']['advancedHandoff'] and all(t['budget'].get('deadlineMs',0)>t['now'] for t in r['trace'][2:]))
check('advanced expired-control control keeps exact identity and no new INSERT',r['writes']==1 and r['first']['transactionId']==r['retry']['transactionId']==r['durable']['transaction_id'] and r['retained']['consumerRetained'])
r=read('round4-advanced-retirement.json')['results'][0]
check('B4-1b explicit consumer retirement preserves advanced recovery obligation',r['after']['writes']==1 and r['after']['reads']>r['before']['reads'])
check('advanced retirement control keeps exact INSERT key and durable COMMIT unchanged',r['before']['durable']==r['after']['durable'] and r['writeTrace'][0]['mutation']==r['writeTrace'][1]['mutation'] and r['writeTrace'][0]['idempotencyKey']==r['writeTrace'][1]['idempotencyKey'])
r=x['live-unresolved-creation-background-visibility']
check('B4-2 existing sweep reconciles live unknown outcome when evidence returns',r['beforeRetry']['rowInserted'] and r['beforeRetry']['reads']>r['before']['reads'])
check('live unknown positive control same-session retry sees returned evidence',r['accepted'] and r['retry']['transactionId']==r['first']['transactionId'])
for r in read('round4-timer-orderings.json')['results']:
 check('B4-3 '+r['order']+' timer then uncertain completion retains live entitlement',r['afterTimer']['consumerRetained'] and r['afterCompletion']['consumerRetained'] and r['accepted'])
 check('timer '+r['order']+' control retains original server debt after uncertain completion',r['afterTimer']['diagnostics']['recordCount']==r['afterCompletion']['diagnostics']['recordCount']==1 and r['afterCompletion']['diagnostics']['operationCount']==0)
r=read('round4-complete-resource-probe.json')['observed']
check('B4-4 envelope covers observed persistence projection and original budget',r['diagnostics']['recordBytes']>=r['minCanonicalBytes'])
check('resource witness observes distinct live roots without VM layout assumptions',r['originalAndTurnBudgetsDistinct'] and r['projectionIsDistinct'] and not r['rawEqualsNormalized'])
for label,ok in checks:print(('PASS ' if ok else 'FAIL ')+label)
print(f'{sum(ok for _,ok in checks)}/{len(checks)} extension assertions pass; {sum(not ok for _,ok in checks)} failures in four grouped findings')
sys.exit(any(not ok for _,ok in checks))
