import {pathToFileURL} from 'node:url';
import inspector from 'node:inspector';
const root=process.argv[2],imp=p=>import(pathToFileURL(root+'/'+p));
const {SQLQueryEngine}=await imp('src/query/sql-query-engine.js');
const silent={debug(){},info(){},warn(){},error(){}};
const session=new inspector.Session();session.connect();
const post=(method,params={})=>new Promise((resolve,reject)=>session.post(method,params,(err,value)=>err?reject(err):resolve(value)));
await post('Debugger.enable');
const fileUrl=pathToFileURL(root+'/src/query/sql-query-engine-transaction-persistence-methods.js').href;
await post('Debugger.setBreakpointByUrl',{url:fileUrl,lineNumber:306});
const pauses=[];let e;
session.on('Debugger.paused',({params})=>{
 const frame=params.callFrames.find(f=>f.functionName==='readDistributedTransactionCreationRow');
 if(!frame){session.post('Debugger.resume');return;}
 const expression=`JSON.stringify((()=>{const count=o=>{let strings=0,numbers=0,booleans=0,references=0;const fields=[];for(const [key,d] of Object.entries(Object.getOwnPropertyDescriptors(o))){if(!('value'in d))throw Error('accessor');const v=d.value;if(typeof v==='string'){strings+=Buffer.byteLength(v);fields.push({key,type:'string',bytes:Buffer.byteLength(v)});}else if(typeof v==='number'){numbers++;fields.push({key,type:'number'});}else if(typeof v==='boolean')booleans++;else if(v&&typeof v==='object')references++;}return {strings,numbers,booleans,references,fields,bytes:strings+8*(numbers+booleans+references)};};return {inputAndCapturedRowsDistinct:read.rows[0]!==rows[0],inputOwn:count(read.rows[0]),capturedOwn:count(rows[0]),inputAndCapturedArrayDistinct:read.rows!==rows};})())`;
 session.post('Debugger.evaluateOnCallFrame',{callFrameId:frame.callFrameId,expression,returnByValue:true},(error,value)=>{
  pauses.push(error?{error:String(error)}:{source:fileUrl,line:frame.location.lineNumber+1,observation:value.result.value?JSON.parse(value.result.value):value,diagnostics:e.transactionCoordinator.getTransactionCreationDiagnostics()});session.post('Debugger.resume');
 });
});
let initial,writes=0,reads=0;
e=new SQLQueryEngine({logger:silent,nowFn:()=>0,autoStartDistributedTransactionRecovery:false,maxUncertainTransactionCreationCanonicalBytes:300000,controlPlaneSystemTableGateway:{supportsMutationSubmission:()=>true,async submitMutation(m){writes++;initial={...m.row};return {success:false,error:'lost reply'};},async readAuthoritativeRows(){reads++;return {success:true,rows:[{...initial}]};}}});
const first=await e.executeQuery('BEGIN',[],{sessionId:'💾'.repeat(5000)});
await e.shutdown();await post('Debugger.disable');session.disconnect();
console.log(JSON.stringify({root,writes,reads,first,pauses},null,2));
