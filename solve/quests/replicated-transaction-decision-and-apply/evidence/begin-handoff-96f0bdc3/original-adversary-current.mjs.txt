function uncertainError() {
  return Object.assign(new Error('uncertain'), {
    code: 'TRANSACTION_CREATION_UNCERTAIN',
  });
}

class RecordingSocket extends EventEmitter {
  constructor() {
    super();
    this.destroyed = false;
    this.written = [];
  }

  write(value) {
    this.written.push(Buffer.from(value));
    return true;
  }

  destroy() {
    this.destroyed = true;
  }
}

async function invalidAdvancedMutableRow() {
  let attempted;
  const engine = new SQLQueryEngine({
    autoStartDistributedTransactionRecovery: false,
    controlPlaneSystemTableGateway: {
      supportsMutationSubmission: () => true,
      async submitMutation(mutation) {
        attempted ||= {...mutation.row};
        return {success: false, error: 'lost answer'};
      },
      async readAuthoritativeRows() {
        return {success: true, rows: [{...attempted, status: 'NOT_A_STATUS'}]};
      },
    },
  });
  try {
    const result = await engine.executeQuery('BEGIN', [], {sessionId: 'bad-row'});
    return {result, diagnostics:
      engine.transactionCoordinator.getTransactionCreationDiagnostics()};
  } finally {
    await engine.shutdown();
  }
}

async function accessorArrayReadback() {
  let attempted;
  let getterCalls = 0;
  const engine = new SQLQueryEngine({
    autoStartDistributedTransactionRecovery: false,
    controlPlaneSystemTableGateway: {
      supportsMutationSubmission: () => true,
      async submitMutation(mutation) {
        attempted ||= {...mutation.row};
        return {success: false, error: 'lost answer'};
      },
      async readAuthoritativeRows() {
        const rows = new Array(1);
        Object.defineProperty(rows, 0, {
          enumerable: true,
          get() {
            getterCalls += 1;
            return {...attempted};
          },
        });
        return {success: true, rows};
      },
    },
  });
  try {
    const result = await engine.executeQuery('BEGIN', [], {sessionId: 'getter'});
    return {result, getterCalls};
  } finally {
    await engine.shutdown();
  }
}

async function expiredRetry() {
  let now = 0;
  let writes = 0;
  const coordinator = new DistributedTransactionCoordinator({
    now: () => now,
    transactionBudgetMs: 10,
    async persistTransaction() {
      writes += 1;
      if (writes === 1) throw uncertainError();
    },
  });
  const first = await coordinator.begin('expired');
  now = 100;
  const retry = await coordinator.begin('expired');
  return {
    first,
    retry,
    active: coordinator.getTransaction('expired'),
    writes,
  };
}

async function sweepPrefixStarvation() {
  let now = 0;
  const writes = new Map();
  const coordinator = new DistributedTransactionCoordinator({
    now: () => now,
    maxUncertainTransactionCreations: 32,
    maxUncertainCreationSweepPerTurn: 16,
    async persistTransaction(record) {
      writes.set(record.transactionId, (writes.get(record.transactionId) || 0) + 1);
      throw uncertainError();
    },
  });
  for (let index = 0; index < 17; index += 1) {
    await coordinator.begin(`sweep-${index}`);
  }
  const ids = [...coordinator.workflowCoordinator.workflowsById.keys()].sort();
  await coordinator.runRecoverySweep();
  now = 2000;
  await coordinator.runRecoverySweep();
  return {first: writes.get(ids[0]), sixteenth: writes.get(ids[15]),
    seventeenth: writes.get(ids[16])};
}

async function postTagAcceptanceFailure() {
  const engine = new SQLQueryEngine({
    autoStartDistributedTransactionRecovery: false,
    controlPlaneSystemTableGateway: {
      supportsMutationSubmission: () => true,
      submitMutation: async () => ({success: true, completionState: 'applied'}),
    },
  });
  const adapter = new PostgresWireAdapter({
    sqlCore: engine,
    authHandler: {
      authenticate: async ({database}) => ({authenticated: true,
        context: {tenantId: database, principal: 'review', roles: []}}),
      authorizeQuery: () => ({authorized: true}),
    },
  });
  const socket = new RecordingSocket();
  const handler = new PgWireProtocolHandler({adapter, socket,
    logger: {debug() {}, error() {}, info() {}, warn() {}}});
  try {
    await handler._completeAuthentication({sessionId: 'post-tag',
      tenantId: 'default', user: 'review', database: 'default'});
    socket.written.length = 0;
    handler.getSession().enterTransaction = () => {
      throw new Error('local-entry-failed');
    };
    await handler._handleSimpleQuery(Buffer.from('BEGIN\0'));
    const tx = engine.transactionCoordinator.workflowCoordinator
      .workflowsByOwnerKey.get('post-tag');
    return {messageTypes: socket.written.map((entry) =>
      String.fromCharCode(entry[0])), socketDestroyed: socket.destroyed,
    creationState: tx?.transactionCreationState,
    sessionClosed: handler.getSession().isClosed()};
  } finally {
    handler.destroy();
    await engine.shutdown();
  }
}

async function closeBeforeTag() {
  let releaseInsert;
  const insertEntered = new Promise((resolve) => {
    releaseInsert = {entered: resolve, finish: null};
  });
  let finishInsert;
  const insertGate = new Promise((resolve) => {
    finishInsert = resolve;
  });
  const engine = new SQLQueryEngine({
    autoStartDistributedTransactionRecovery: false,
    controlPlaneSystemTableGateway: {
      supportsMutationSubmission: () => true,
      async submitMutation() {
        releaseInsert.entered();
        await insertGate;
        return {success: true, completionState: 'applied'};
      },
    },
  });
  const adapter = new PostgresWireAdapter({
    sqlCore: engine,
    authHandler: {
      authenticate: async ({database}) => ({authenticated: true,
        context: {tenantId: database, principal: 'review', roles: []}}),
      authorizeQuery: () => ({authorized: true}),
    },
  });
  const socket = new RecordingSocket();
  const handler = new PgWireProtocolHandler({adapter, socket,
    logger: {debug() {}, error() {}, info() {}, warn() {}}});
  try {
    await handler._completeAuthentication({sessionId: 'close-before-tag',
      tenantId: 'default', user: 'review', database: 'default'});
    socket.written.length = 0;
    const query = handler._handleSimpleQuery(Buffer.from('BEGIN\0'));
    await insertEntered;
    handler.destroy();
    finishInsert();
    await query;
    const tx = engine.transactionCoordinator.workflowCoordinator
      .workflowsByOwnerKey.get('close-before-tag');
    return {messageTypes: socket.written.map((entry) =>
      String.fromCharCode(entry[0])), socketDestroyed: socket.destroyed,
    creationState: tx?.transactionCreationState,
    sessionClosed: handler.getSession().isClosed()};
  } finally {
    await engine.shutdown();
  }
}

async function accessorConfiguration() {
  let getterCalls = 0;
  const options = {};
  Object.defineProperty(options, 'maxUncertainTransactionCreations', {
    get() {
      getterCalls += 1;
      return 1;
    },
  });
  let accepted = false;
  try {
    new DistributedTransactionCoordinator(options);
    accepted = true;
  } catch {}
  return {accepted, getterCalls};
}

async function shutdownDuringDirectBegin() {
  let releasePersist;
  const enteredPersist = new Promise((resolve) => {
    releasePersist = {entered: resolve, finish: null};
  });
  let finishPersist;
  const persistGate = new Promise((resolve) => {
    finishPersist = resolve;
  });
  const coordinator = new DistributedTransactionCoordinator({
    async persistTransaction() {
      releasePersist.entered();
      await persistGate;
    },
  });
  const beginPromise = coordinator.begin('shutdown-direct');
  await enteredPersist;
  coordinator.retireUncertainCreationsForProcessShutdown();
  finishPersist();
  const result = await beginPromise;
  return {result, active: coordinator.getTransaction('shutdown-direct'),
    diagnostics: coordinator.getTransactionCreationDiagnostics()};
}
