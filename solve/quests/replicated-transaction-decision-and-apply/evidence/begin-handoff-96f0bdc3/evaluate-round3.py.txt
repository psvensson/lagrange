 ok=(r['beforeRetry']['sameRecord'] and r['afterRetry']['sameRecord'] and r['afterRetry']['consumerRetained'] and r['afterRetry']['recordCount']==1 and len(r['commitEmissions'])==1 and not r['rollbackEmissions'] and r['commitEmissions'][0]['identity']['decisionDigest']==digest and r['afterReplay']['durable']['decision_digest']==digest and r['afterReplay']['durable']['decision']=='COMMIT' and not r['finalRetry']['success'] and r['afterFinal']['recordCount']==0)
