// Design/ordering evidence ONLY. No Raft implementation, TX1 model or D2 oracle.
// Explicit event schedules assume ordered-prefix commit and an atomic applier.
// Native observations separately support only existing Ready/apply primitives.
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
const digest = x => createHash('sha256').update(JSON.stringify(x)).digest('hex');
const copy = x => structuredClone(x);
const mutant = process.argv.includes('--allow-retire-overtake');
const outputs = [];
const events = [];
let serial = 0;
const a = (g, nonce, version = 0, valid = true) => {
  const body = ['APPROVE', 'p', ['1', '1'], String(g), nonce, String(version), valid];
  return {kind: 'A', g, version, valid, body, id: digest(body)};
};
const r = (g, nonce, version = 0, valid = true) => {
  const child = a(g + 1, nonce, version, valid);
  const body = ['RETIRE_AND_APPROVE', 'p', ['1', '1'], String(g), child.body];
  return {kind: 'R', g, child, body, id: digest(body)};
};
const t = g => {
  const body = ['RETIRE', 'p', ['1', '1'], String(g)];
  return {kind: 'T', g, body, id: digest(body)};
};
class Projection {
  constructor() {
    this.db = {g: 0, version: 0, mode: 'HELD', results: new Map(), receipt: null,
      applied: 0, authorityEffects: 0, authorityK: null};
    this.log = [];
    this.commit = 0;
    this.live = new Map();
    this.fenced = false;
    this.logEvents = [];
  }
  event(name, details = {}) { this.logEvents.push({name, ...details}); }
  inventory() {
    const occurrences = [...this.live.values()];
    const ordinary = occurrences.filter(x => x.cmd.kind === 'A');
    const futureA = new Set(ordinary.filter(x => !this.db.results.has(x.cmd.id)).map(x => x.cmd.id));
    const maintenance = occurrences.filter(x => ['R', 'T'].includes(x.cmd.kind));
    const futureM = new Set(maintenance.filter(x => x.cmd.kind === 'R' &&
      !this.db.results.has(x.cmd.child.id) && this.db.receipt?.id !== x.cmd.id).map(x => x.cmd.id));
    return {qA: ordinary.length, qM: maintenance.length,
      retainedA: this.db.results.size, futureA: futureA.size, futureM: futureM.size,
      aEntitlements: this.db.results.size + futureA.size};
  }
  read(cmd) {
    if (cmd.kind === 'A' && this.db.results.has(cmd.id)) return copy(this.db.results.get(cmd.id));
    if (cmd.kind !== 'A' && this.db.receipt?.id === cmd.id) return copy(this.db.receipt);
    if (cmd.g < this.db.g) return {status: 'HISTORY_RETIRED', disposition: 'UNAVAILABLE'};
    return {status: 'UNRESOLVED'};
  }
  admit(cmd, path = 'local') {
    assert.equal(digest(cmd.body), cmd.id, 'canonical identity checked');
    if (cmd.kind === 'R') {
      assert.equal(cmd.child.g, cmd.g + 1);
      assert.deepEqual(cmd.body.at(-1), cmd.child.body);
      assert.equal(digest(cmd.child.body), cmd.child.id);
    }
    const result = this.read(cmd);
    if (result.status !== 'UNRESOLVED') return {status: 'RESULT', result};
    if (this.fenced) return {status: 'NOT_ADMITTED', reason: 'RECONSTRUCTION'};
    const exact = [...this.live.values()].find(x => x.cmd.id === cmd.id);
    if (exact) {
      assert.deepEqual(exact.cmd.body, cmd.body);
      exact.waiters = Math.min(2, exact.waiters + 1);
      this.event('join', {id: cmd.id, path});
      return {status: 'JOIN', ticket: exact.ticket};
    }
    const inv = this.inventory();
    if (inv.qA > 4 || inv.qM > 1) return {status: 'NOT_ADMITTED', reason: 'OVER_PROFILE'};
    if (cmd.g !== this.db.g) return {status: 'NOT_ADMITTED', reason: 'GENERATION'};
    if (cmd.kind === 'A' && (inv.qA >= 4 || inv.aEntitlements >= 8))
      return {status: 'NOT_ADMITTED', reason: 'A_CAPACITY'};
    if (cmd.kind !== 'A' && inv.qM && !(mutant && cmd.kind === 'T'))
      return {status: 'NOT_ADMITTED', reason: 'MAINTENANCE_BUSY'};
    const ticket = ++serial;
    this.live.set(ticket, {ticket, cmd: copy(cmd), phase: 'pre-core', index: null, waiters: 1});
    this.event('admit', {id: cmd.id, kind: cmd.kind, ticket, path, inventory: this.inventory()});
    return {status: 'ADMITTED', ticket};
  }
  append(ticket) {
    const q = this.live.get(ticket);
    assert.ok(q && q.phase === 'pre-core');
    assert.equal(this.fenced, false);
    q.phase = 'core';
    q.index = this.log.length + 1; // Actual order event, never submit time.
    this.log.push({index: q.index, term: 1, ticket, cmd: copy(q.cmd)});
    this.event('ordered', {ticket, index: q.index});
    return q.index;
  }
  phase(ticket, phase) {
    const q = this.live.get(ticket);
    const phases = ['core', 'ready', 'durable'];
    assert.equal(phases.indexOf(phase), phases.indexOf(q.phase) + 1);
    q.phase = phase;
    this.event(phase, {ticket, index: q.index});
  }
  timeout(ticket) {
    const q = this.live.get(ticket);
    const before = this.inventory();
    q.waiters = 0;
    assert.deepEqual(this.inventory(), before);
    this.event('waiter-timeout-debt-retained', {ticket, inventory: before});
  }
  commitThrough(index) {
    assert.ok(index >= this.commit && index <= this.log.length);
    for (const e of this.log.filter(x => x.index > this.commit && x.index <= index))
      assert.equal(this.live.get(e.ticket).phase, 'durable');
    this.commit = index;
    this.event('commit-prefix', {index});
  }
  applyNext({failTransaction = false} = {}) {
    const e = this.log[this.db.applied];
    assert.ok(e && e.index <= this.commit);
    const before = copy(this.db);
    const cmd = e.cmd;
    let disposition;
    try {
      if (cmd.kind === 'A') {
        if (this.db.results.has(cmd.id)) disposition = this.db.results.get(cmd.id).disposition;
        else if (cmd.g < this.db.g) disposition = 'HISTORY_RETIRED';
        else {
          assert.equal(cmd.g, this.db.g);
          disposition = this.applyChild(cmd, e);
        }
      } else if (this.db.receipt?.id === cmd.id) disposition = 'REPLAY';
      else if (cmd.g < this.db.g) disposition = 'HISTORY_RETIRED';
      else {
        assert.equal(cmd.g, this.db.g);
        this.db.g++;
        // Policy assumption: no obligation is pinned only by a release-result row.
        // TX/query history does not exist in this narrow projection and is not deleted.
        this.db.results.clear();
        disposition = cmd.kind === 'R' ? this.applyChild(cmd.child, e) : null;
        this.db.receipt = {status: 'RETIRED', id: cmd.id, fromG: cmd.g,
          toG: this.db.g, index: e.index, term: e.term,
          childK: cmd.child?.id ?? null, childDisposition: disposition};
      }
      this.db.applied = e.index;
      if (failTransaction) throw new Error('explicit environmental failure');
    } catch (error) {
      this.db = before;
      if (!failTransaction) throw error;
      this.event('atomic-rollback', {index: e.index});
      return {status: 'ENVIRONMENT_FAILURE'};
    }
    this.live.delete(e.ticket); // After atomic exchange; all other physical copies remain.
    this.event('applied', {index: e.index, g: this.db.g, disposition, inventory: this.inventory()});
    assert.ok(this.inventory().aEntitlements <= 8, 'result exchange within conditional A bound');
    return {status: 'APPLIED', disposition, index: e.index};
  }
  applyChild(cmd, entry) {
    assert.equal(cmd.g, this.db.g);
    // Symbolic D2 predicate ONLY; this cannot validate real release provenance/discharge.
    const ok = cmd.valid && cmd.version === this.db.version && this.db.mode === 'HELD';
    const disposition = ok ? 'APPROVED' : 'REFUSED';
    const result = {status: 'RESULT', disposition, id: cmd.id, body: copy(cmd.body),
      index: entry.index, term: entry.term, winner: ok ? cmd.id : this.db.authorityK};
    this.db.results.set(cmd.id, result);
    if (ok) { this.db.version++; this.db.mode = 'OPEN'; this.db.authorityEffects++; this.db.authorityK = cmd.id; }
    return disposition;
  }
  finish(ticket) {
    this.append(ticket); this.phase(ticket, 'ready'); this.phase(ticket, 'durable');
    this.commitThrough(this.log.length);
    while (this.db.applied < this.commit) this.applyNext();
  }
  loseLeader() { this.fenced = true; this.event('leader-loss'); }
  reconcile({discardFrom = null, provenPreCore = []} = {}) {
    // These inputs represent the REQUIRED authoritative runtime/store interface,
    // not an interface supplied by current production code or a map-miss guess.
    if (discardFrom !== null) {
      assert.ok(discardFrom > this.commit, 'cannot discard committed prefix');
      for (const e of this.log.filter(x => x.index >= discardFrom)) this.live.delete(e.ticket);
      this.log = this.log.filter(x => x.index < discardFrom);
    }
    for (const ticket of provenPreCore) {
      assert.equal(this.live.get(ticket)?.phase, 'pre-core');
      this.live.delete(ticket);
    }
    for (const e of this.log.filter(x => x.index > this.db.applied)) assert.ok(this.live.has(e.ticket));
    this.fenced = false;
    this.event('authoritative-reconstruction', {discardFrom, provenPreCore, inventory: this.inventory()});
  }
  recoveredDuplicate(ticket) {
    const original = this.live.get(ticket);
    assert.ok(original.index !== null);
    const next = ++serial;
    const index = this.log.length + 1;
    const duplicate = {...copy(original), ticket: next, index, phase: 'durable'};
    this.live.set(next, duplicate);
    this.log.push({index, term: 1, ticket: next, cmd: copy(original.cmd)});
    this.event('recovered-extra-physical-copy', {index, inventory: this.inventory()});
    return next;
  }
}
function run(name, work) {
  const s = new Projection();
  work(s);
  outputs.push({name, status: 'PASS'});
  events.push({name, events: s.logEvents, final: {g: s.db.g, version: s.db.version,
    applied: s.db.applied, effects: s.db.authorityEffects, inventory: s.inventory()}});
}
function fill(s, n = 8) {
  for (let i = 0; i < n; i++) {
    const q = s.admit(a(s.db.g, `refusal-${s.db.g}-${i}`, s.db.version, false));
    assert.equal(q.status, 'ADMITTED'); s.finish(q.ticket);
  }
}
