// W07 test-only witness. The frozen schedule and classification are recorded
// in native-runtime-service-w07-20261015/WITNESS-PLAN.md. No release command
// is modeled here. The service endpoint is actual native propose entry, not
// proposal acceptance, forwarding, commitment or application.
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {test} from 'node:test';

import {VirtualTimeSource} from '../../../src/time/time-source.js';
import {RAFT_MEMBERSHIP_OPERATION, RAFT_OPERATION_OUTCOME} from
  '../../../src/raft/raft-operation-port-constants.js';
import {RAFT_RS_MESSAGE_TYPE} from '../../../src/raft/raft-rs-ingress-constants.js';
import {encodeProposal} from '../../../src/raft/raft-rs-proposal-codec.js';
import {RUNTIME_PHASE} from '../../../src/raft/raft-rs-runtime-owner-constants.js';
import {setActualCoreEntryObserver, setCoreFaultInjector,
  setNativeInboundStageObserver} from '../../../src/raft/raft-rs-runtime-owner.js';
import {PARTITION_TIMING, PartitionNodeCluster} from './partition-node-cluster.js';

const REPLICAS = ['w07-a', 'w07-b', 'w07-c'];
const [A, B, C] = REPLICAS;
const ARRIVALS = 4;
const SERVICE_BOUND = 1;
const MICROTASK_GUARD = 256;
const SETUP_ROUNDS = 12;
const CLOCK_START = 1000;

function hash(value) {
  return createHash('sha256').update(JSON.stringify(value)).digest('hex');
}

function deferred() {
  let resolve;
  const promise = new Promise((done) => {
    resolve = done;
  });
  return {promise, resolve};
}

// This runs no clock, port operation or transport helper. Failure here is a
// setup/seam failure, never the service-bound RED. The fixed guard only lets
// already-scheduled Promise continuations run; it cannot enter Raft itself.
async function observe(predicate, label) {
  for (let turn = 0; turn < MICROTASK_GUARD; turn += 1) {
    if (predicate()) return;
    await Promise.resolve();
  }
  assert.fail(`seam guard: ${label}`);
}

function processSetupInbox(cluster, replica) {
  for (const envelope of cluster.replica(replica).inbox.splice(0)) {
    cluster.node(replica).step(envelope);
  }
  return cluster.node(replica).readStatus();
}

function form(cluster) {
  assert.equal(cluster.settle(() => cluster.leaderReplicaId() === A), true);
  cluster.propose(A, {setup: cluster.partitionId});
  assert.equal(cluster.settle(() => REPLICAS.every((id) =>
    cluster.replica(id).appliedCommands.length === 1)), true);
  for (let round = 0; round < SETUP_ROUNDS; round += 1) {
    if (REPLICAS.every((id) => cluster.replica(id).inbox.length === 0)) return;
    for (const id of REPLICAS) processSetupInbox(cluster, id);
  }
  assert.fail('setup: RF3 transport did not quiesce');
}

// Capture an unmodified actual A -> B heartbeat. Manual ticks and status
// reads below are SETUP ONLY. Other native outputs remain in fixture inboxes.
function captureHeartbeat(cluster) {
  for (let turn = 0; turn < SETUP_ROUNDS; turn += 1) {
    cluster.node(A).tick();
    const inbox = cluster.replica(B).inbox;
    const at = inbox.findIndex((item) =>
      item.message.msgType === RAFT_RS_MESSAGE_TYPE.HEARTBEAT);
    if (at !== -1) return inbox.splice(at, 1)[0];
  }
  assert.fail('setup: no native heartbeat captured');
}

function fixture(name) {
  const events = [];
  const gates = [];
  const clocks = new Map(REPLICAS.map((id) =>
    [id, new VirtualTimeSource({startMs: CLOCK_START})]));
  const control = {hold: null, capture: false};
  const cluster = new PartitionNodeCluster({
    partitionId: name, replicaIds: REPLICAS,
    substrateFor: (id) => ({timeSource: clocks.get(id)}),
    sendFor: (from, address, packet) => {
      if (from !== B) return undefined;
      if (control.capture) {
        events.push({kind: 'native-send', hash: hash(packet),
          message: packet.message});
      }
      if (!control.hold?.(packet)) return undefined;
      const gate = deferred();
      const ordinal = gates.length + 1;
      gates.push({...gate, packet, address});
      events.push({kind: 'native-send-held', ordinal: gates.length,
        hash: hash(packet), message: packet.message});
      return gate.promise.then(() => {
        cluster.queue(from, address, packet);
        events.push({kind: 'native-send-released', ordinal});
      });
    },
  });
  form(cluster);
  const heartbeat = captureHeartbeat(cluster);
  const peer = cluster.raftPeerIdOf(B);
  // Flush only the owned zero-time SETUP callbacks, while no measured input
  // or operation exists. No election clock is started on this RF3 fixture.
  for (const clock of clocks.values()) clock.advance(0);
  return {cluster, events, gates, clocks, control, heartbeat, peer};
}

// These callbacks only copy events. The fault seam supplies the args omitted
// by actualCoreEntryObserver, paired with that observer's sequence/generation.
function installObservers(f, fault = () => undefined) {
  let entry = null;
  setActualCoreEntryObserver((observation) => {
    entry = observation;
    f.events.push({kind: 'core-entry', ...observation});
  });
  setCoreFaultInjector((groupId, operation, args) => {
    if (groupId === f.cluster.partitionId && operation === 'step') {
      f.events.push({kind: 'core-step', ...entry,
        hash: hash(args[0]), message: args[0]});
    }
    if (groupId === f.cluster.partitionId && operation === 'propose') {
      f.events.push({kind: 'core-propose', ...entry,
        bytes: Buffer.from(args[0]).toString('hex')});
    }
    fault(groupId, operation);
  });
  setNativeInboundStageObserver((event) => {
    if (event.groupId === f.cluster.partitionId) {
      f.events.push({kind: 'inbound-stage', ...event});
    }
  });
}

function deliver(f, ordinal) {
  const envelopeHash = hash(f.heartbeat);
  const result = f.cluster.node(B).step(f.heartbeat);
  f.events.push({kind: 'delivery', ordinal,
    provenance: ordinal === 1 ? 'captured-native-heartbeat' :
      'deliberate-exact-duplicate', envelopeHash, result});
  assert.equal(result.outcome, RAFT_OPERATION_OUTCOME.CORE_OK);
  assert.equal(result.reason, 'inbound-enqueued');
  assert.equal(hash(f.heartbeat), envelopeHash, 'delivery does not change bytes');
}

function proposalEntry(f, command) {
  const bytes = Buffer.from(encodeProposal(command)).toString('hex');
  return f.events.find((event) => event.kind === 'core-propose' && event.bytes === bytes);
}

function steps(f) {
  return f.events.filter((event) => event.kind === 'core-step' &&
    event.message.to === f.peer);
}

function finish(t, f, extra = {}) {
  t.diagnostic(JSON.stringify({w07: f.cluster.partitionId, ...extra,
    events: f.events}));
  setCoreFaultInjector(null);
  setActualCoreEntryObserver(null);
  setNativeInboundStageObserver(null);
  for (const gate of f.gates) gate.resolve();
  f.cluster.dispose();
}

async function serviceCase(t, arrivals) {
  const f = fixture(`w07-service-${arrivals}`);
  const predecessor = {w07: 'already-owned-forwarded-proposal'};
  const selected = {w07: 'queued-ordinary-proposal'};
  try {
    installObservers(f);
    f.control.capture = true;
    f.control.hold = (packet) => (f.gates.length === 0 &&
      packet.message.msgType === RAFT_RS_MESSAGE_TYPE.PROPOSE) ||
      packet.message.msgType === RAFT_RS_MESSAGE_TYPE.HEARTBEAT_RESPONSE;
    const first = f.cluster.node(B).propose(predecessor);
    await observe(() => f.gates.length === 1, 'predecessor native forwarding is held');
    assert.ok(proposalEntry(f, predecessor));
    f.events.push({kind: 'measurement-cut', arrivals, serviceBound: SERVICE_BOUND,
      predecessor: 'core-entered, Ready awaits its native forwarded send'});
    const queued = f.cluster.node(B).propose(selected);
    if (arrivals > 0) deliver(f, 1);
    f.gates[0].resolve();
    await first;
    for (let ordinal = 1; ordinal <= arrivals; ordinal += 1) {
      // Only the B-owned zero-time inbound wake is dispatched. No tick,
      // readStatus, settle or deliverAll participates in this interval.
      f.clocks.get(B).advance(0);
      await observe(() => f.gates.filter((gate) => gate.packet.message.msgType ===
        RAFT_RS_MESSAGE_TYPE.HEARTBEAT_RESPONSE).length >= ordinal,
      `native heartbeat reply ${ordinal}`);
      if (ordinal < arrivals) deliver(f, ordinal + 1);
      const replies = f.gates.filter((gate) => gate.packet.message.msgType ===
        RAFT_RS_MESSAGE_TYPE.HEARTBEAT_RESPONSE);
      replies[ordinal - 1].resolve();
    }
    await observe(() => Boolean(proposalEntry(f, selected)), 'selected native propose entry');
    const result = await queued;
    f.events.push({kind: 'measurement-end', result});
    assert.equal(result.outcome, RAFT_OPERATION_OUTCOME.CORE_OK);
    assert.equal(f.events.filter((event) => event.kind === 'native-send' &&
      event.message.msgType === RAFT_RS_MESSAGE_TYPE.PROPOSE).length, 2,
    'both callable proposals generated actual forwarded native outputs');
    const entered = steps(f);
    assert.equal(entered.length, arrivals, 'no accepted remainder disappears');
    assert.deepEqual(entered.map((event) => event.hash),
      Array(arrivals).fill(hash(f.heartbeat.message)), 'each occurrence retains exact bytes');
    assert.equal(f.events.filter((event) => event.kind === 'inbound-stage' &&
      event.stage === 'accepted' && event.replicaIdentity === B).length, arrivals);
    assert.equal(f.gates.filter((gate) => gate.packet.message.msgType ===
      RAFT_RS_MESSAGE_TYPE.HEARTBEAT_RESPONSE).length, arrivals,
    'each stepped heartbeat produces its real native reply');
    const before = entered.filter((event) =>
      event.sequence < proposalEntry(f, selected).sequence).length;
    f.events.push({kind: 'service-observation', inboundBeforeSelected: before,
      bound: SERVICE_BOUND, preservation: 'passed'});
    assert.ok(before <= SERVICE_BOUND,
      `semantic service RED: ${before} post-cut inbound steps precede selected propose; bound ${
        SERVICE_BOUND}`);
  } finally {
    finish(t, f);
  }
}

test('W07 quiet input: existing queued proposal completes', (t) => serviceCase(t, 0));
test('W07 finite recurring inbound: preserve all bytes and service captured round',
  (t) => serviceCase(t, ARRIVALS));

test('W07 replacement: suspended invocation cannot enter the replacement generation', async (t) => {
  const f = fixture('w07-generation');
  const triggerId = 'w07-replacement-trigger';
  const triggerClock = new VirtualTimeSource({startMs: CLOCK_START});
  const trigger = new PartitionNodeCluster({partitionId: triggerId, replicaIds: [triggerId],
    substrateFor: () => ({timeSource: triggerClock})});
  const selected = {w07: 'must-not-enter-after-replacement'};
  let trapped = false;
  try {
    trigger.node(triggerId).startScheduling();
    installObservers(f, (groupId, operation) => {
      if (groupId === triggerId && operation === 'tick' && !trapped) {
        trapped = true;
        f.events.push({kind: 'declared-one-shot-native-trap'});
        throw new globalThis.WebAssembly.RuntimeError('W07 declared shared runtime replacement');
      }
    });
    f.control.hold = (packet) => f.gates.length === 0 &&
      packet.message.msgType === RAFT_RS_MESSAGE_TYPE.HEARTBEAT_RESPONSE;
    deliver(f, 1);
    const pending = f.cluster.node(B).propose(selected);
    await observe(() => f.gates.length === 1, 'old generation native response held');
    const originalGeneration = steps(f)[0].runtimeGeneration;
    deliver(f, 2);
    f.events.push({kind: 'measurement-cut', originalGeneration,
      ownedRemainder: hash(f.heartbeat), pending: selected});
    // Two explicitly declared owned-clock ticks: first injects the native
    // trap, second invokes the existing shared reconstruction owner.
    triggerClock.advance(PARTITION_TIMING.tickIntervalMs);
    assert.equal(trapped, true);
    triggerClock.advance(PARTITION_TIMING.tickIntervalMs);
    assert.ok(f.events.some((event) => event.kind === 'core-entry' &&
      event.operation === 'create_node' && event.runtimeGeneration > originalGeneration),
    'seam: the actual shared runtime reconstructed before resumption');
    const resumeAt = f.events.length;
    f.gates[0].resolve();
    const result = await pending;
    assert.equal(f.events.slice(resumeAt).some((event) => event.kind === 'core-entry' &&
      event.groupId === f.cluster.partitionId), false,
    'stale continuation enters no target primitive before the new owned-clock turn');
    f.events.push({kind: 'old-invocation-result', result});
    assert.equal(result.outcome, RAFT_OPERATION_OUTCOME.HOST_FAILURE);
    assert.equal(result.phase, RUNTIME_PHASE.GENERATION_CHANGED);
    assert.equal(proposalEntry(f, selected), undefined);
    f.clocks.get(B).advance(0);
    await observe(() => steps(f).length === 2, 'accepted remainder serviced after reconstruction');
    assert.deepEqual(steps(f).map((event) => event.hash),
      [hash(f.heartbeat.message), hash(f.heartbeat.message)]);
    assert.ok(steps(f)[1].runtimeGeneration > originalGeneration,
      'only the reconstructed runtime enters the preserved remainder');
    f.events.push({kind: 'measurement-end', preservedRemainder: true});
  } finally {
    trigger.node(triggerId).stopScheduling();
    finish(t, f);
    trigger.dispose();
  }
});

test('W07 refusal: a native per-message refusal preserves the preaccepted tail', async (t) => {
  const f = fixture('w07-refusal');
  try {
    // Re-quiesce the setup heartbeat before changing membership.
    for (const id of REPLICAS) processSetupInbox(f.cluster, id);
    f.cluster.node(A).proposeConfChange({type: RAFT_MEMBERSHIP_OPERATION.REMOVE_PEER,
      replicaIdentity: C});
    processSetupInbox(f.cluster, B);
    processSetupInbox(f.cluster, C);
    const acknowledgements = f.cluster.replica(A).inbox.splice(0).filter((envelope) =>
      envelope.message.msgType === RAFT_RS_MESSAGE_TYPE.APPEND_RESPONSE);
    const cPeer = f.cluster.raftPeerIdOf(C);
    const refused = acknowledgements.find((envelope) => envelope.message.from === cPeer);
    const accepted = acknowledgements.find((envelope) => envelope.message.from !== cPeer);
    assert.ok(refused && accepted, 'setup: actual B/C removal acknowledgements');
    f.cluster.node(A).step(accepted);
    f.cluster.node(A).readStatus();
    await f.cluster.node(B).propose({w07: 'preaccepted-tail'});
    const forwarded = f.cluster.replica(A).inbox.splice(0).find((envelope) =>
      envelope.message.msgType === RAFT_RS_MESSAGE_TYPE.PROPOSE);
    assert.ok(forwarded, 'setup: actual follower proposal');
    installObservers(f);
    const selected = {w07: 'after-refused-and-valid-preaccepted'};
    f.events.push({kind: 'measurement-cut', nativeOrigin: [refused, forwarded]});
    for (const envelope of [refused, forwarded]) {
      assert.equal(f.cluster.node(A).step(envelope).outcome, RAFT_OPERATION_OUTCOME.CORE_OK);
    }
    const result = await f.cluster.node(A).propose(selected);
    f.events.push({kind: 'measurement-end', result});
    assert.equal(result.outcome, RAFT_OPERATION_OUTCOME.CORE_OK);
    const entered = f.events.filter((event) => event.kind === 'core-step');
    assert.deepEqual(entered.map((event) => event.hash),
      [hash(refused.message), hash(forwarded.message)]);
    assert.ok(entered.every((event) => event.sequence < proposalEntry(f, selected).sequence),
      'both preaccepted messages precede the command, preserving CA1/M9');
    // Ended interval: both steps and the selected command already entered.
    // This diagnostic read cannot rescue their measured service.
    const status = f.cluster.node(A).readStatus();
    const refusal = status.inboundStepRefusals.find((event) => event.from === cPeer);
    assert.equal(refusal?.outcome, RAFT_OPERATION_OUTCOME.CORE_REFUSED);
    assert.equal(refusal?.phase, 'step');
    f.events.push({kind: 'post-measurement-refusal', refusal});
  } finally {
    finish(t, f);
  }
});
