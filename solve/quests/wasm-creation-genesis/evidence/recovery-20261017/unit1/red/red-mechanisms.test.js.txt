// Red-run evidence only (scratchpad; never committed): the design's named red
// mechanisms observed on the base 381750de2 with the shimmed fixture.
import {test} from 'node:test';
import assert from 'node:assert/strict';
import {embedding} from './bound-creation-handler-embedding-fixture.js';
import {ReplicaOperationMessageType as TYPE} from '../../src/rebalancer/replica-operation-constants.js';

test('BASE P-R1 mechanism: after a host restart a REMOVE of a retained creation is NOT_FOUND', async (t) => {
  const e = await embedding(t);
  for (let slot = 0; slot < 3; slot += 1) await e.addHost(slot);
  await e.retain(e.hosts[1]);
  const restarted = await e.restartHost(e.hosts[1]);
  const reply = await e.dispatch(restarted, TYPE.REMOVE_REPLICA, 'remove-base');
  t.diagnostic(JSON.stringify(reply));
  assert.equal(reply.status, 'initiated', `BASE answers ${reply.status}`);
});

test('BASE P-R12 mechanism: a REMOVE of a PENDING row with an applied reservation orphans the ADMITTED slot', async (t) => {
  const e = await embedding(t);
  for (let slot = 0; slot < 3; slot += 1) await e.addHost(slot);
  const host = e.hosts[1];
  e.metadata.loseReservation();
  await e.retain(host);
  e.metadata.loseReservation();
  await e.dispatch(host, TYPE.CREATE_REPLICA);
  await e.settle(() => host.outcomes.length === 1);
  await e.dispatch(host, TYPE.REMOVE_REPLICA, 'remove-pending');
  await e.settle(() => host.outcomes.length === 2);
  const phase = e.metadata.state().selected.resources[1].phase;
  t.diagnostic(JSON.stringify({outcome: host.outcomes[1].outcomeType, phase,
    row: e.metadata.db.prepare('SELECT state FROM wasm_operations WHERE operation_id = ?').get(host.intent.consensusSlotRequest.operationId)}));
  assert.equal(phase, 'CLOSED', `BASE leaves the slot ${phase}`);
});
