

// ---- VERIFIER COVERAGE PROBES (temporary; never committed; restored byte-identically) ----
test('PX1 (verifier probe) a close that loses to another caller release of the SAME slot re-decides to CLOSED',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const [token] = await reservedSlots(f, binding, 1);
    assert.equal((await f.primary.grantInstanceResource(binding, token)).outcome, 'PUBLISHED');
    const from = f.writes.length;
    const held = f.holdNext('owner-a');
    t.after(held.release);
    const pending = f.primary.closeInstanceResource(binding, token);
    await held.entered;
    assert.equal((await f.owner('owner-b').definitions.releaseInstanceResource(binding, token))
      .outcome, 'RELEASING');
    held.release();
    assert.deepEqual(await pending, {outcome: 'CLOSED', phase: 'CLOSED', token}, 'PX1_CLOSE_NOT_RE_DECIDED');
    assert.deepEqual(updatesSince(f, 'owner-a', from), [0, 1], 'PX1_WRITES');
    assert.equal(f.state().selected.resources[0].phase, 'CLOSED');
  });

test('PX2 (verifier probe) an absent affected-row count is UNKNOWN after one submission, never re-decided',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const [token] = await reservedSlots(f, binding, 1);
    const gateway = f.primary.getGateway();
    const update = gateway.updateSystemTableRow.bind(gateway);
    let submissions = 0;
    gateway.updateSystemTableRow = async (...args) => {
      submissions += 1;
      if (submissions > 1) return update(...args);
      return {success: true};
    };
    const before = f.row();
    assert.deepEqual(await f.primary.grantInstanceResource(binding, token), {outcome: 'UNKNOWN'},
      'PX2_AN_ABSENT_COUNT_WAS_NOT_UNKNOWN');
    assert.equal(submissions, 1, 'PX2_AN_ABSENT_COUNT_WAS_RESUBMITTED');
    assert.deepEqual(f.row(), before);
  });

test('PX3 (verifier probe) a reserve that loses the SAME slot to another operation re-decides to CONFLICT',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const from = f.writes.length;
    const held = f.holdNext('owner-a');
    t.after(held.release);
    const pending = f.primary.reserveInstanceResource(binding, f.slotRequest(0, 0, 'first', 0));
    await held.entered;
    const winner = admitted(await f.owner('owner-b').definitions.reserveInstanceResource(binding,
      f.slotRequest(0, 0, 'second', 1)));
    held.release();
    const loser = await pending;
    assert.equal(loser.outcome, 'CONFLICT', JSON.stringify(loser));
    assert.deepEqual(loser.winnerToken, winner);
    assert.deepEqual(updatesSince(f, 'owner-a', from), [0], 'PX3_WRITES');
    assert.equal(f.state().selected.resources[0].request.operationId, 'second');
  });

test('PX4 (verifier probe) a release that loses to a sibling slot grant re-decides to RELEASING',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const [mine, sibling] = await reservedSlots(f, binding, 2);
    assert.equal((await f.primary.grantInstanceResource(binding, mine)).outcome, 'PUBLISHED');
    const from = f.writes.length;
    const held = f.holdNext('owner-a');
    t.after(held.release);
    const pending = f.primary.releaseInstanceResource(binding, mine);
    await held.entered;
    assert.equal((await f.owner('owner-b').definitions.grantInstanceResource(binding, sibling))
      .outcome, 'PUBLISHED');
    held.release();
    assert.deepEqual(await pending, {outcome: 'RELEASING', phase: 'RELEASING', token: mine},
      'PX4_RELEASE_NOT_RE_DECIDED');
    assert.deepEqual(updatesSince(f, 'owner-a', from), [0, 1], 'PX4_WRITES');
    assert.equal(f.state().selected.resources[1].phase, 'PUBLISHED');
  });

test('PX5 (verifier probe) bound exhaustion costs RESOURCES submissions and RESOURCES+1 authoritative reads',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const [token] = await reservedSlots(f, binding, 1);
    const gateway = f.primary.getGateway();
    const read = gateway.readAuthoritativeRows;
    let reads = 0;
    let submissions = 0;
    gateway.readAuthoritativeRows = async (...args) => {
      reads += 1;
      return read(...args);
    };
    gateway.updateSystemTableRow = async () => {
      submissions += 1;
      return {success: true, affectedRows: 0};
    };
    assert.deepEqual(await f.primary.grantInstanceResource(binding, token), {outcome: 'STATE_CHANGED'});
    assert.equal(submissions, SERVICE_INSTANCE_LIMIT.RESOURCES, 'PX5_SUBMISSIONS');
    assert.equal(reads, SERVICE_INSTANCE_LIMIT.RESOURCES + 1, `PX5_READS ${reads}`);
  });

test('PX6 (verifier probe) a sealed change landing after the last submission answers the sealed refusal, not STATE_CHANGED',
  async (t) => {
    const f = await createCreationSqliteFixture(t);
    const binding = selected(await f.primary.selectConsensusInstance(f.request()));
    const [token] = await reservedSlots(f, binding, 1);
    const gateway = f.primary.getGateway();
    let submissions = 0;
    gateway.updateSystemTableRow = async () => {
      submissions += 1;
      if (submissions === SERVICE_INSTANCE_LIMIT.RESOURCES) {
        selected(await f.owner('owner-b').definitions.selectConsensusInstance(
          f.request(1, 2, binding.incarnation)));
      }
      return {success: true, affectedRows: 0};
    };
    const answer = await f.primary.grantInstanceResource(binding, token);
    assert.equal(submissions, SERVICE_INSTANCE_LIMIT.RESOURCES, 'PX6_SUBMISSIONS');
    assert.deepEqual(answer, {outcome: 'STALE_BINDING'}, `PX6_ANSWER ${JSON.stringify(answer)}`);
  });
