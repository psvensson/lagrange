def sha(b): return hashlib.sha256(b).hexdigest()
def read(p): return open(os.path.join(ROOT, p), 'rb').read()
def write(p, b): open(os.path.join(ROOT, p), 'wb').write(b)

names = sys.argv[1:] or list(M)
summary = {}
for name in names:
    edits, files = M[name]
    originals = {}
    for path, _, _ in edits:
        originals.setdefault(path, read(path))
    texts = {p: b.decode() for p, b in originals.items()}
    for path, old, new in edits:
        if old is None:
            texts[path] = texts[path] + new
        else:
            assert texts[path].count(old) == 1, (name, path, old)
            texts[path] = texts[path].replace(old, new)
    log = open(f'{OUT}/{name}.log', 'w')
    res = {}
    try:
        for path, text in texts.items():
            write(path, text.encode())
        for f in files:
            g = subprocess.run(['node', 'scripts/checks/wait-for-thermal-headroom.js'], cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
            log.write(g.stdout.decode(errors='replace'))
            r = subprocess.run(['npm', 'run', 'test:file', '--', f], cwd=ROOT, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, timeout=900)
            log.write(r.stdout.decode(errors='replace'))
            tap = f'{ROOT}/.tap/test-results/{f}.tap'
            fails = subprocess.run(['python3', '-I', TAPFAIL, tap], stdout=subprocess.PIPE).stdout.decode()
            log.write(f'\n==== {f} exit {r.returncode}\n{fails}\n')
            res[f] = {'exit': r.returncode, 'failures': fails}
    finally:
        restored = {}
        for path, original in originals.items():
            write(path, original)
            restored[path] = sha(read(path)) == sha(original)
            log.write(f'\nRESTORED {path} sha256 {sha(original)} byte-identical={restored[path]}\n')
        log.close()
        assert all(restored.values()), (name, restored)
    summary[name] = res
    print('=====', name)
    for f, v in res.items():
        print(' ', f, 'exit', v['exit'])
        print('   ' + v['failures'].replace('\n', '\n   ')[:2500])
    sys.stdout.flush()
json.dump(summary, open(f'{OUT}/summary-{"-".join(names) if sys.argv[1:] else "all"}.json', 'w'), indent=1)
