// Read-only release audit witness; no Lagrange source is modified.
// Real parser/planner/write coordinator/rendering and native SQLite; the
// partition transport alone is replaced by a recording local SQLite sink.
import Database from '../../lagrange-review/node_modules/better-sqlite3/lib/index.js';
import {ConfigurationManager} from '../../lagrange-review/src/config/configuration-manager.js';
import {SQLQueryEngine} from '../../lagrange-review/src/query/sql-query-engine.js';
import {SQLParser} from '../../lagrange-review/src/query/sql-parser.js';
import {TableCreationService} from '../../lagrange-review/src/query/table-creation-service.js';
import {createPartitionServiceTable} from '../../lagrange-review/src/partition/partition-service-table-bootstrap.js';
import {validateUserTableCreateIntent} from '../../lagrange-review/src/query/user-table-create-intent.js';
import {createMockSystemCache, createMockMessageRouter} from '../../lagrange-review/test/query/sql-query-engine-test-support.js';

const quiet = {debug(){}, info(){}, warn(){}, error(){}};
ConfigurationManager.getInstance().initialize();
const db = new Database(':memory:');
db.pragma('foreign_keys = ON');
db.exec('CREATE TABLE p (id INTEGER PRIMARY KEY)');
db.exec('CREATE TABLE fk_control (id INTEGER PRIMARY KEY, parent INTEGER REFERENCES p(id))');
let declaredForeignKeyRefused = false;
try { db.exec('INSERT INTO fk_control VALUES(1,999)'); } catch (error) { declaredForeignKeyRefused = error.code === 'SQLITE_CONSTRAINT_FOREIGNKEY'; }
console.log(JSON.stringify({kind:'foreign-key-control', foreignKeys:db.pragma('foreign_keys',{simple:true}), declaredForeignKeyRefused}));
const emittedDdl = [];
function createTable(original) {
  const ast = new SQLParser(original, {dialect:'postgresql'}).parse();
  validateUserTableCreateIntent(ast);
  const schema = TableCreationService.prototype.buildSchemaDefinition(ast.columns);
  const service = {tableName:ast.tableName, schema, logger:quiet,
    db:{exec(sql){emittedDdl.push(sql); db.exec(sql);}}};
  for (const name of ['ensureServicesTableColumns','ensureTablesTableColumns',
    'ensureMessageGroupsTableColumns','ensurePartitionsTableColumns',
    'ensureSqlTransactionsTableColumns','ensureReplicaOperationsTableColumns']) {
    service[name] = () => {};
  }
  createPartitionServiceTable(service);
  return {original, emittedDdl:emittedDdl.splice(0),
    tableInfo:db.prepare(`PRAGMA table_info(${ast.tableName})`).all(),
    indexes:db.prepare(`PRAGMA index_list(${ast.tableName})`).all()};
}

console.log(JSON.stringify({kind:'environment', node:process.version,
  sqlite:db.prepare('SELECT sqlite_version() AS version').get().version}));
createTable('CREATE TABLE t (id INTEGER PRIMARY KEY, amount INTEGER, txt TEXT)');
db.prepare('INSERT INTO t VALUES (1, 41, NULL)').run();
const engine = new SQLQueryEngine({
  systemCache:createMockSystemCache([{table_name:'t', primaryKey:'id'}],
    [{partition_id:'p1',table_name:'t',partition_key_start:null,partition_key_end:null}]),
  messageRouter:createMockMessageRouter(), autoStartDistributedTransactionRecovery:false,
});
for (const obj of [engine,engine.queryExecutor,engine.distributedWriteCoordinator,
  engine.distributedQueryPlanner,engine.partitionResolver]) if(obj)obj.logger=quiet;
const sent = [];
const execute = (partitionId, sql, params) => {
  sent.push({partitionId,sql,params});
  const result=db.prepare(sql).run(...params);
  return {partitionId,success:true,changes:result.changes,rows:[]};
};
engine.queryExecutor.executeOnPartition = async (...args) => execute(...args);
engine.queryExecutor.executeOnPartitions = async (partitionIds,sql,params) =>
  partitionIds.map(id=>execute(id,sql,params));
for (const original of ['UPDATE t SET amount = amount + 1 WHERE id = 1',
  'INSERT INTO t(id, amount) VALUES (2, 2 + 3)',
  'INSERT INTO t(id, txt) VALUES (3, $$hello$$)',
  'UPDATE t SET amount = CAST(5 AS INTEGER) WHERE id = 1',
  "UPDATE t SET txt = UPPER('hi') WHERE id = 1"]) {
  const result = await engine.executeQuery(original, [], {dialect:'postgresql'});
  console.log(JSON.stringify({kind:'mutation', original,
    success:result.success,affectedRows:result.affectedRows,sent:sent.splice(0),
    storedRows:db.prepare('SELECT * FROM t ORDER BY id').all()}));
}
const cases = [
  {sql:'CREATE TABLE t_composite (id INTEGER, name TEXT, PRIMARY KEY(id,name))', insert:"INSERT INTO t_composite VALUES (1,'a'),(1,'a')",table:'t_composite'},
  {sql:'CREATE TABLE t_unique (id INTEGER PRIMARY KEY, email TEXT UNIQUE)',
    insert:"INSERT INTO t_unique VALUES (1,'same'),(2,'same')", table:'t_unique'},
  {sql:'CREATE TABLE t_pk (id INTEGER, amount INTEGER, PRIMARY KEY(id), UNIQUE(amount))',
    insert:'INSERT INTO t_pk VALUES (1,10),(1,10)',table:'t_pk'},
  {sql:'CREATE TABLE t_check (id INTEGER PRIMARY KEY, amount INTEGER CHECK(amount>=0))',
    insert:'INSERT INTO t_check VALUES (1,-100)',table:'t_check'},
  {sql:'CREATE TABLE t_fk (id INTEGER PRIMARY KEY, parent INTEGER REFERENCES p(id))',
    insert:'INSERT INTO t_fk VALUES (1,999)',table:'t_fk'},
];
for (const entry of cases) {
  const result=createTable(entry.sql);
  let violationAccepted=false;
  let error=null;
  try {db.exec(entry.insert);violationAccepted=true;}catch(e){error=e.message;}
  console.log(JSON.stringify({kind:'constraint',...result,insert:entry.insert,
    violationAccepted,error,storedRows:db.prepare(`SELECT * FROM ${entry.table}`).all()}));
}
engine.transactionCoordinator.stopRecoverySweep?.();
db.close();
