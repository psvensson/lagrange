import fs from 'node:fs';
const [a,b]=process.argv.slice(2).map(f=>JSON.parse(fs.readFileSync(f,'utf8')));
const out=[];
function key(arr){return arr.every(x=>x&&typeof x==='object'&&('id' in x||'item' in x))?(x=>x.id??('item'+x.item)):null}
function d(x,y,p){
 if(JSON.stringify(x)===JSON.stringify(y))return;
 if(Array.isArray(x)&&Array.isArray(y)){const k=key([...x,...y]);
   if(k&&x.length+y.length>0){const mx=new Map(x.map(e=>[k(e),e])),my=new Map(y.map(e=>[k(e),e]));
     for(const [i,e] of mx){if(!my.has(i))out.push(['REMOVED',p+'['+i+']',JSON.stringify(e).slice(0,200)]);else d(e,my.get(i),p+'['+i+']');}
     for(const [i,e] of my)if(!mx.has(i))out.push(['ADDED',p+'['+i+']',JSON.stringify(e).slice(0,200)]);
     if(JSON.stringify([...mx.keys()])!==JSON.stringify([...my.keys()].filter(i=>mx.has(i))))out.push(['ORDER',p,'']);
     return;}
   const n=Math.max(x.length,y.length);for(let i=0;i<n;i++)d(x[i],y[i],p+'['+i+']');return;}
 if(x&&y&&typeof x==='object'&&typeof y==='object'&&!Array.isArray(x)&&!Array.isArray(y)){
   for(const k of new Set([...Object.keys(x),...Object.keys(y)])){
     if(!(k in x))out.push(['ADDED',p+'.'+k,JSON.stringify(y[k]).slice(0,300)]);
     else if(!(k in y))out.push(['REMOVED',p+'.'+k,JSON.stringify(x[k]).slice(0,300)]);
     else d(x[k],y[k],p+'.'+k);}
   return;}
 out.push(['CHANGED',p,JSON.stringify(x)?.slice(0,300)+'  =>  '+JSON.stringify(y)?.slice(0,300)]);
}
d(a,b,'');
for(const o of out)console.log(o.join(' | '));
console.log('TOTAL',out.length);
