import fs from 'node:fs';
const S='/mnt/data/peter/projects/lagrange/.claude/worktrees/audit-evidence-binding/solve/epics/formation-seed-decoupling/overflow-budget-decision-matrix.json';
const a=JSON.parse(fs.readFileSync('inh/matrix.json','utf8')), b=JSON.parse(fs.readFileSync(S,'utf8'));
const out=[];
function byId(arr){return Array.isArray(arr)&&arr.length&&arr.every(x=>x&&typeof x==='object'&&('id'in x||'item'in x))}
function walk(x,y,p){
  if(JSON.stringify(x)===JSON.stringify(y))return;
  if(byId(x)&&byId(y)){const k=e=>e.id??e.item; const mx=new Map(x.map(e=>[k(e),e])),my=new Map(y.map(e=>[k(e),e]));
    for(const [i,e] of mx){ if(!my.has(i))out.push(['REMOVED',p+'['+i+']',e]); else walk(e,my.get(i),p+'['+i+']'); }
    for(const [i,e] of my) if(!mx.has(i))out.push(['ADDED',p+'['+i+']',e]);
    if(JSON.stringify(x.map(k))!==JSON.stringify(y.map(k)) && x.length===y.length) out.push(['ORDER?',p,'']); return;}
  if(x&&y&&typeof x==='object'&&typeof y==='object'&&!Array.isArray(x)&&!Array.isArray(y)){
    for(const k of new Set([...Object.keys(x),...Object.keys(y)])){ if(!(k in y))out.push(['REMOVED',p+'.'+k,x[k]]); else if(!(k in x))out.push(['ADDED',p+'.'+k,y[k]]); else walk(x[k],y[k],p+'.'+k);} return;}
  out.push(['CHANGED',p,x,y]);
}
walk(a,b,'');
const cls={};
for(const o of out){ const key=o[0]+' '+o[1].replace(/\[[^\]]+\]/g,'[*]'); (cls[key]=cls[key]||[]).push(o); }
for(const [k,v] of Object.entries(cls)){ console.log(v.length+'x',k); if(v.length<=3||/CHANGED/.test(k)) for(const o of v) console.log('     ',o[1],'::',JSON.stringify(o[2]).slice(0,160),o[3]!==undefined?'=> '+JSON.stringify(o[3]).slice(0,160):''); }
console.log('total paths',out.length);
