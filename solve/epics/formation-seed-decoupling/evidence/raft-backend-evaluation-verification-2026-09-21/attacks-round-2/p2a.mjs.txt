import {createDeterministicCluster, confChangeV2, loadForkedCore} from './forked-core-harness.js';
const core=loadForkedCore(); const enc=(s)=>Buffer.from(s,'utf8');
const fatalMsg=(to)=>({from:'2',to,term:'1',logTerm:'0',index:'0',msgType:8,entries:[],commit:'99',reject:false,rejectHint:'0'});
function fullScenario(tag){ const c=createDeterministicCluster({voters:['1','2','3'],learners:['4']}); try{
  c.core.campaign(c.handleOf('1')); c.settle(400); const h=c.handleOf('1');
  c.core.propose(h,enc('a')); c.core.propose_conf_change_v2(h,confChangeV2([{type:1,nodeId:'2'},{type:0,nodeId:'4'}],2)); c.core.propose(h,enc('b')); c.settle(400);
  c.core.propose_conf_change_v2(h,confChangeV2([])); c.settle(400);
  c.crash('3'); c.restart('3'); c.tickPeer('1',5); c.settle(400);
  const cs=[...c.confStates()].filter(([id])=>id!=='2').map(([id,s])=>`${s.voters}|${s.votersOutgoing}|${s.learners}`); const ok=new Set(cs).size===1&&cs[0]==='1,3,4||';
  const errs=[...c.peers.keys()].flatMap(id=>c.applyConfChangeErrors(id));
  if(!ok||errs.length) console.log('BYSTANDER MISBEHAVED',tag,cs,JSON.stringify(errs)); return ok&&!errs.length; } finally{c.free();} }
console.log('baseline ok',fullScenario('baseline'),'handles',core.handle_count(),'mem',core.wasm_memory_bytes());

const by=createDeterministicCluster({voters:['1','2','3'],learners:['4']}); by.core.campaign(by.handleOf('1')); by.settle(400);
let trapped=0, other=0, firstOther=null; const mem0=core.wasm_memory_bytes(); const N=Number(process.argv[2]||3000);
for(let i=0;i<N;i++){
  const c=createDeterministicCluster({voters:['1','2','3'],learners:[]}); 
  try{ c.core.step(c.handleOf('2'),fatalMsg('2')); other++; }catch(e){ if(/unreachable/.test(String(e.message)))trapped++; else {other++; firstOther ||= String(e.message)+' @'+i;} }
  try{ c.free(); }catch(e){ firstOther ||= 'free: '+e.message; }
  if(i%500===499){ by.core.propose(by.handleOf('1'),enc('k'+i)); by.settle(400); const st=by.statuses(); const commits=[...st.values()].map(s=>String(s.commit)); console.log('after',i+1,'fatals trapped',trapped,'other',other,firstOther,'mem',core.wasm_memory_bytes(),'handles',core.handle_count(),'bystander commits',commits.join(','), 'full ok', fullScenario('after '+(i+1))); }
}
console.log('mem growth',core.wasm_memory_bytes()-mem0,'per fatal',(core.wasm_memory_bytes()-mem0)/N);
// 1000 create/free cycles after the fatals
let cf=0; for(let i=0;i<1000;i++){ const h=core.create_node({id:'1',peers:['1','2','3'],learners:[],electionTick:10,heartbeatTick:3}); core.tick(h); core.free(h); cf++; }
console.log('create/free cycles',cf,'handles',core.handle_count(),'mem',core.wasm_memory_bytes());
// fatal inside create_node (applied ahead of log) x 200
let cn=0, cnMsg=null; for(let i=0;i<200;i++){ try{ core.create_node({id:'1',peers:['1'],learners:[],electionTick:10,heartbeatTick:3,applied:'50',bootstrap:{entries:[],confState:{voters:['1'],learners:[],votersOutgoing:[],learnersNext:[],autoLeave:false}}}); }catch(e){cn++; cnMsg=String(e.message);} }
console.log('create_node fatals',cn,cnMsg,'handles',core.handle_count(),'full ok',fullScenario('after create fatals'));

const d=createDeterministicCluster({voters:['1','2','3'],learners:[]}); const dh=d.handleOf('2'); try{d.core.step(dh,fatalMsg('2'));}catch(e){}
for(const [n,f] of [['has_ready dead',()=>core.has_ready(dh)],['free dead',()=>core.free(dh)],['free unknown',()=>core.free(999999)],['status dead',()=>core.status(dh)],['tick dead',()=>core.tick(dh)]]){ try{console.log(n,'->',JSON.stringify(f()));}catch(e){console.log(n,'threw',String(e.message));} }
