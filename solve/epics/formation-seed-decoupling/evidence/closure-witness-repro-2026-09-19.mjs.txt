const svc=(p,r,node,role,status='active')=>({service_id:`${p}-${r}`,service_type:'partition',partition_id:p,node_id:node,status,raft_role:role,address:`ws://${node}:1`,updated_at:NOW-1000});

// 4 voters on TWO nodes (n0,n1) for every critical partition; learner on n2.
function rows({learner}) {
  const out=[];
  for (const p of CRIT) {
    out.push(svc(p,'r1','n0','LEADER'),svc(p,'r2','n0','FOLLOWER'),svc(p,'r3','n1','FOLLOWER'),svc(p,'r4','n1','FOLLOWER'));
    if (learner) out.push(svc(p,'r5','n2',LEARNER_ROLE,LEARNER_STATUS));
  }
  return out;
}
